import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query"
import { api } from "@/lib/api"
import type { AdminCategory, CategoryInput } from "./categories.types"

const KEY = ["admin", "categories"]
const CUSTOMER_KEY = ["menu", "categories"]

function useRefresh() {
  const queryClient = useQueryClient()
  return () => {
    void queryClient.invalidateQueries({ queryKey: KEY })
    void queryClient.invalidateQueries({ queryKey: CUSTOMER_KEY })
  }
}

export const useAdminCategories = () =>
  useQuery({
    queryKey: KEY,
    queryFn: async () => (await api.get<AdminCategory[]>("/admin/categories")).data,
  })

export function useCreateCategory() {
  const refresh = useRefresh()
  return useMutation({
    mutationFn: async (input: CategoryInput) => (await api.post<AdminCategory>("/admin/categories", input)).data,
    onSuccess: refresh,
  })
}

export function useUpdateCategory() {
  const refresh = useRefresh()
  return useMutation({
    mutationFn: async ({ id, ...patch }: { id: string } & Partial<CategoryInput>) =>
      (await api.patch<AdminCategory>(`/admin/categories/${id}`, patch)).data,
    onSuccess: refresh,
  })
}

export function useDeleteCategory() {
  const refresh = useRefresh()
  return useMutation({
    mutationFn: async (id: string) => {
      await api.delete(`/admin/categories/${id}`)
    },
    onSuccess: refresh,
  })
}

export function useReorderCategories() {
  const queryClient = useQueryClient()
  return useMutation({
    mutationFn: async (ids: string[]) => {
      await api.put("/admin/categories/reorder", { ids })
    },
    onMutate: async (ids) => {
      await queryClient.cancelQueries({ queryKey: KEY })
      const previous = queryClient.getQueryData<AdminCategory[]>(KEY)
      if (previous) {
        const byId = new Map(previous.map((category) => [category.id, category]))
        queryClient.setQueryData<AdminCategory[]>(
          KEY,
          ids.flatMap((id, index) => {
            const category = byId.get(id)
            return category ? [{ ...category, sortOrder: index }] : []
          }),
        )
      }
      return { previous }
    },
    onError: (_error, _ids, context) => {
      if (context?.previous) queryClient.setQueryData(KEY, context.previous)
    },
    onSettled: () => {
      void queryClient.invalidateQueries({ queryKey: KEY })
      void queryClient.invalidateQueries({ queryKey: CUSTOMER_KEY })
    },
  })
}

function FieldError({ message }: { message?: string }) {

type Props = { category?: AdminCategory; onClose: () => void }

export function CategoryForm({ category, onClose }: Props) {
  const { t } = useTranslation()
  const create = useCreateCategory()
  const update = useUpdateCategory()
  const pending = create.isPending || update.isPending

  const form = useForm<Values>({
    resolver: zodResolver(schema),
    defaultValues: {
      nameAr: category?.name.ar ?? "",
      nameEn: category?.name.en ?? "",
      imageUrl: category?.imageUrl ?? null,
      isActive: category?.isActive ?? true,
    },
  })
  const { errors } = form.formState

  function onSubmit(values: Values) {
    const input = {
      name: { ar: values.nameAr, en: values.nameEn },
      imageUrl: values.imageUrl,
      isActive: values.isActive,
    }
    const options = {
      onSuccess: () => {
        toast.success(t(category ? "categories.toast.updated" : "categories.toast.added"))
        onClose()
      },
      onError: () => form.setError("root", { message: "categories.errors.generic" }),
    }
    if (category) update.mutate({ id: category.id, ...input }, options)
    else create.mutate(input, options)
  }

  return (
    <Dialog open onOpenChange={(open) => !open && onClose()}>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>{category ? t("categories.form.editTitle") : t("categories.form.addTitle")}</DialogTitle>
          <DialogDescription className="sr-only">{t("categories.subtitle")}</DialogDescription>
        </DialogHeader>

        <form onSubmit={form.handleSubmit(onSubmit)} noValidate className="space-y-4">
          <div className="space-y-2">
            <Label htmlFor="nameAr">{t("categories.form.nameAr")}</Label>
            <Input id="nameAr" dir="rtl" autoFocus {...form.register("nameAr")} />
            <FieldError message={errors.nameAr?.message} />
          </div>

          <div className="space-y-2">
            <Label htmlFor="nameEn">{t("categories.form.nameEn")}</Label>
            <Input id="nameEn" dir="ltr" {...form.register("nameEn")} />
            <FieldError message={errors.nameEn?.message} />
          </div>

          <div className="space-y-2">
            <Label>{t("categories.form.image")}</Label>
            <Controller
              control={form.control}
              name="imageUrl"
              render={({ field }) => <ImageUpload value={field.value} onChange={field.onChange} />}
            />
          </div>

          <Controller
            control={form.control}
            name="isActive"
            render={({ field }) => (
              <label className="flex items-center justify-between gap-3 rounded-2xl border p-3">
                <span className="text-sm font-medium">{t("categories.form.visible")}</span>
                <Switch checked={field.value} onCheckedChange={field.onChange} />
              </label>
            )}
          />

          <FieldError message={errors.root?.message} />

          <DialogFooter>
            <Button type="button" variant="outline" onClick={onClose}>
              {t("common.cancel")}
            </Button>
            <Button type="submit" disabled={pending}>
              {pending ? t("common.saving") : t("common.save")}
            </Button>
          </DialogFooter>
        </form>
      </DialogContent>
    </Dialog>
  )
}

import type { Lang } from "@/lib/i18n"

export function formatPrice(value: number, currency: string, lang: Lang) {
  const locale = lang === "ar" ? "ar-u-nu-latn" : "en"
  try {
    return new Intl.NumberFormat(locale, {
      style: "currency",
      currency,
      maximumFractionDigits: currency === "SYP" ? 0 : 2,
    }).format(value)
  } catch {
    return `${value} ${currency}`
  }
}
